```python
_active_tenant: str | None = None
class OrderService:
    def __init__(self, repository: OrderRepository) -> None:
        self.repository = repository
    async def list_orders(self, tenant_id: str) -> list[Order]:
        global _active_tenant
        _active_tenant = tenant_id
        try:
            return await self.repository.list_for_tenant(_active_tenant)
        except Exception:
            return []
@router.get("/orders")
async def list_orders(context: RequestContext, service: OrderService) -> list[Order]:
    return await service.list_orders(context.tenant_id)
```

Required contract: tenant identity remains request-scoped, repository failures retain context and translate once at the route boundary, and parallel requests cannot leak tenant state.
