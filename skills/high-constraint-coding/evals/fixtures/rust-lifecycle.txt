pub struct JobRunner { lease: LeaseGuard, client: Client }
impl JobRunner {
    pub async fn run(&mut self, cancelled: CancellationToken) -> Result<Job, Error> {
        let lease_tx = self.lease.sender();
        let heartbeat = tokio::spawn(async move {
            loop { lease_tx.renew().await?; tokio::time::sleep(Duration::from_secs(5)).await; }
        });
        tokio::select! {
            result = self.client.next_job() => result,
            _ = cancelled.cancelled() => return Err(Error::Cancelled),
        }
    }
}
