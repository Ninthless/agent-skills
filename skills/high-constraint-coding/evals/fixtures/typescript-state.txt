type CheckoutState =
  | { status: "idle" }
  | { status: "submitting" }
  | { status: "succeeded"; receipt: Receipt }
  | { status: "failed"; error: CheckoutError };
type CheckoutPatch = { isLoading: boolean; receipt?: Receipt; error?: string };
export function renderCheckout(state: CheckoutPatch) {
  if (state.isLoading) return "Submitting";
  if (state.receipt) return state.receipt!.number;
  return state.error ?? "Ready";
}
