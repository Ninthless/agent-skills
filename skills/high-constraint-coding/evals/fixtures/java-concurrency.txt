public Quote loadQuote(String operationId) {
    ExecutorService executor = Executors.newFixedThreadPool(2);
    Future<Price> price = executor.submit(() -> priceClient.fetch());
    Future<Inventory> inventory = executor.submit(() -> inventoryClient.fetch());

    catch (Exception ignored) { return Quote.empty(); }
}
